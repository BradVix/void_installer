#!/bin/bash
# VOID LINUX 2025 – ABSOLUTELY FINAL PERFECT INSTALLER
# Now with sudo/doas choice!

set -e
clear
echo "=== VOID LINUX 2025 – FINAL PERFECT INSTALLER ==="
echo "Now with sudo vs doas choice"
read -p "Press Enter to begin…"

# ——— WiFi in live environment ———
echo -e "\nWiFi setup (needed for repos)"
read -p "Interface (wlan0/wlp2s0/etc): " WIFI_IFACE
read -p "SSID: " SSID
read -s -p "Password: " WIFI_PASS; echo
ip link set "$WIFI_IFACE" up
wpa_passphrase "$SSID" "$WIFI_PASS" > /tmp/wpa.conf
wpa_supplicant -B -i "$WIFI_IFACE" -c /tmp/wpa.conf
dhcpcd "$WIFI_IFACE" &
sleep 10
ping -c 1 repo-default.voidlinux.org &>/dev/null || { echo "No internet!"; exit 1; }

# ——— Disk & partitions ———
read -p "Target disk (/dev/sda or /dev/nvme0n1): " DISK
read -p "Launch cfdisk now? (y/n): " P; [[ $P == y* ]] && cfdisk "$DISK"
[ -d /sys/firmware/efi ] && UEFI=1 || UEFI=0
[ $UEFI = 1 ] && read -p "EFI partition: " EFI_PART
read -p "Root partition: " ROOT_PART
read -p "Swap partition (optional): " SWAP_PART

# ——— Options ———
read -p "Full disk encryption? (y/N): " ENCRYPT; ENCRYPT=${ENCRYPT,,}
read -p "GRUB password? (y/N): " GRUBPASS; GRUBPASS=${GRUBPASS,,}

echo -e "\nDesktop environment:"
echo "1) i3   2) KDE Plasma   3) Hyprland   4) None"
read -p "Choice [4]: " DE; DE=${DE:-4}

echo -e "\nPrivilege escalator:"
echo "1) sudo   2) doas"
read -p "Choice [1]: " PRIV; PRIV=${PRIV:-1}

# ——— Encryption & formatting ———
if [[ $ENCRYPT == y ]]; then
    cryptsetup luksFormat --type luks2 "$ROOT_PART"
    cryptsetup open "$ROOT_PART" cryptroot
    ROOT_DEV="/dev/mapper/cryptroot"
    mkfs.ext4 -F "$ROOT_DEV"
    LUKS_UUID=$(blkid -s UUID -o value "$ROOT_PART")
else
    ROOT_DEV="$ROOT_PART"
    mkfs.ext4 -F "$ROOT_DEV"
fi
[ $UEFI = 1 ] && mkfs.vfat -F32 "$EFI_PART"
[ -n "$SWAP_PART" ] && mkswap "$SWAP_PART" && swapon "$SWAP_PART"

mount "$ROOT_DEV" /mnt
[ $UEFI = 1 ] && mkdir -p /mnt/boot/efi && mount "$EFI_PART" /mnt/boot/efi

# ——— Base packages ———
COMMON="base-system base-devel cryptsetup bash nano vim htop curl wget git iw wireless_tools dbus elogind seatd polkit pipewire wireplumber pipewire-pulse easyeffects linux linux-firmware xdg-desktop-portal-hyprland xdg-desktop-portal-wlr xdg-desktop-portal-gtk"

case $DE in
1) DE_PKGS="i3 i3status dmenu xorg xinit terminus-font" ;;
2) DE_PKGS="kde-plasma sddm konsole NetworkManager" ;;
3) DE_PKGS="Hyprland kitty waybar wofi mako rofi-wayland swww" ;;
*) DE_PKGS="" ;;
esac

case $PRIV in
2) PRIV_PKGS="doas" ; PRIV_CFG="echo 'permit persist :wheel' > /etc/doas.conf" ;;
*) PRIV_PKGS="sudo" ; PRIV_CFG="echo '%wheel ALL=(ALL:ALL) ALL' > /etc/sudoers.d/10-wheel" ;;
esac

xbps-install -S -r /mnt -R "https://repo-default.voidlinux.org/current" $COMMON $DE_PKGS $PRIV_PKGS

genfstab -U /mnt >> /mnt/etc/fstab

# ——— Final chroot script ———
cat > /mnt/install-final.sh <<EOF
#!/bin/bash
set -e
export WIFI_IFACE="$WIFI_IFACE" SSID="$SSID" WIFI_PASS="$WIFI_PASS" UEFI=$UEFI ENCRYPT="$ENCRYPT" GRUBPASS="$GRUBPASS" DE=$DE PRIV=$PRIV
[ -n "$LUKS_UUID" ] && export LUKS_UUID="$LUKS_UUID"

# Hostname & hosts
read -p "Hostname [void]: " HOSTNAME; HOSTNAME=\${HOSTNAME:-void}
echo "\$HOSTNAME" > /etc/hostname
cat > /etc/hosts <<EOS
127.0.0.1       localhost
::1             localhost
127.0.1.1       \$HOSTNAME.localdomain \$HOSTNAME
EOS

# Timezone & locale
read -p "Timezone (Europe/Berlin): " TZ
ln -sf /usr/share/zoneinfo/\$TZ /etc/localtime
echo "en_US.UTF-8 UTF-8" >> /etc/default/libc-locales
xbps-reconfigure -f glibc-locales

# Bash forever
sed -i 's|^SHELL=.*|SHELL=/bin/bash|' /etc/default/useradd
chsh -s /bin/bash root

# Root password
passwd

# User
read -p "Create user? (y/n): " CU
if [[ \$CU == y* ]]; then
    read -p "Username: " USER
    useradd -m -G wheel,seat,audio,video,input \$USER
    passwd \$USER
    $PRIV_CFG
    chmod 440 /etc/sudoers.d/10-wheel 2>/dev/null || true
fi

# Encryption
[[ \$ENCRYPT == y ]] && echo "cryptroot UUID=\$LUKS_UUID none luks,discard" > /etc/crypttab

# WiFi / networking
if [ \$DE = 2 ]; then
    nmcli con add type wifi ifname "\$WIFI_IFACE" con-name "\$SSID" ssid "\$SSID" wifi-sec.key-mgmt wpa-psk wifi-sec.psk "\$WIFI_PASS"
    nmcli con mod "\$SSID" connection.autoconnect yes
    xbps-remove -Ryy wpa_supplicant dhcpcd iw wireless_tools
else
    mkdir -p /etc/wpa_supplicant
    wpa_passphrase "\$SSID" "\$WIFI_PASS" > "/etc/wpa_supplicant/wpa_supplicant-\$WIFI_IFACE.conf"
fi

# Services
ln -s /etc/sv/dbus /var/service/
ln -s /etc/sv/elogind /var/service/
ln -s /etc/sv/seatd /var/service/
ln -s /etc/sv/pipewire /var/service/
ln -s /etc/sv/wireplumber /var/service/
ln -s /etc/sv/pipewire-pulse /var/service/
[[ \$DE != 2 ]] && ln -s /etc/sv/{wpa_supplicant,dhcpcd} /var/service/
[[ \$DE = 2 ]] && ln -s /etc/sv/{NetworkManager,sddm} /var/service/

# seatd config
echo 'SEATD_GROUPS=wheel,seat' > /etc/sv/seatd/config

# GRUB password
[[ \$GRUBPASS == y ]] && grub-mkpasswd-pbkdf2 | awk '/hash/{print "password_pbkdf2 root " $NF}' > /etc/grub.d/40_password

# Bootloader
if [ \$UEFI = 1 ]; then
    xbps-install -y grub-x86_64-efi
    grub-install --target=x86_64-efi --efi-directory=/boot/efi --bootloader-id=Void
else
    grub-install "$DISK"
fi

# GRUB cmdline
if [[ \$ENCRYPT == y ]]; then
    sed -i "s|GRUB_CMDLINE_LINUX_DEFAULT=.*|GRUB_CMDLINE_LINUX_DEFAULT=\"loglevel=4 rd.luks.uuid=\$LUKS_UUID root=/dev/mapper/cryptroot\"|" /etc/default/grub
fi

xbps-reconfigure -fa
echo -e "\nINSTALLATION COMPLETE! You can now reboot."
EOF

chmod +x /mnt/install-final.sh
xchroot /mnt /install-final.sh
rm /mnt/install-final.sh

umount -R /mnt
[ -n "$SWAP_PART" ] && swapoff "$SWAP_PART"
echo -e "\nAll done! Reboot and enjoy your perfect Void Linux system."
